  user_id uuid [pk]
  username varchar [unique, not null]
  email varchar [unique, not null]
  password_hash text [not null]
  full_name varchar
  is_active boolean
  is_verified boolean
  created_at timestamp
  updated_at timestamp
  last_login_at timestamp
}
